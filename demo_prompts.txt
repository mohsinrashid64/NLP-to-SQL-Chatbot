============================================================
FIND ASSEMBLY LINES WITH DATA (run this SQL first)
============================================================
SELECT t.TestUnitName, COUNT(*) as nok_count
FROM _tblpartinspentry p
JOIN tbltestunit t ON p.TestUnitID = t.TestUnitID
WHERE p.PartState = 0
GROUP BY t.TestUnitName
ORDER BY nok_count DESC
LIMIT 10;
